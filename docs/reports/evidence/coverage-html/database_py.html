1"""SQLite persistence with transactions, constraints and online backups.""" 

2import os 

3import sqlite3 

4from contextlib import contextmanager 

5from pathlib import Path 

6DB_PATH = os.environ.get('ACADEMIC_DB_PATH', str(Path(__file__).with_name('academic.db')))

8@contextmanager 

9def get_conn(): 

10 conn = sqlite3.connect(DB_PATH, timeout=15) 

11 conn.row_factory = sqlite3.Row 

12 conn.execute('PRAGMA foreign_keys=ON') 

13 try: 

14 with conn: 

15 yield conn 

16 finally: 

17 conn.close()

19def init_db(): 

20 with get_conn() as conn: 

21 conn.executescript(''' 

22 CREATE TABLE IF NOT EXISTS students ( 

23 student_id INTEGER PRIMARY KEY, name TEXT NOT NULL, program TEXT, 

24 semester INTEGER CHECK(semester BETWEEN 1 AND 12)); 

25 CREATE TABLE IF NOT EXISTS subjects ( 

26 subject_id INTEGER PRIMARY KEY, code TEXT NOT NULL, name TEXT NOT NULL); 

27 CREATE TABLE IF NOT EXISTS academic_records ( 

28 record_id INTEGER PRIMARY KEY AUTOINCREMENT, 

29 student_id INTEGER NOT NULL REFERENCES students(student_id), 

30 subject_id INTEGER NOT NULL REFERENCES subjects(subject_id), period TEXT NOT NULL, 

31 attendance_pct REAL NOT NULL CHECK(attendance_pct BETWEEN 0 AND 100), 

32 assignment_score REAL NOT NULL CHECK(assignment_score BETWEEN 0 AND 100), 

33 internal_score REAL NOT NULL CHECK(internal_score BETWEEN 0 AND 100), 

34 exam_score REAL NOT NULL CHECK(exam_score BETWEEN 0 AND 100)); 

35 CREATE TABLE IF NOT EXISTS users ( 

36 username TEXT PRIMARY KEY, password_hash TEXT NOT NULL, 

37 role TEXT NOT NULL CHECK(role IN ('Admin','Faculty','Student')), 

38 student_id INTEGER REFERENCES students(student_id), active INTEGER NOT NULL DEFAULT 1, 

39 CHECK(role != 'Student' OR student_id IS NOT NULL)); 

40 CREATE TABLE IF NOT EXISTS audit_log ( 

41 id INTEGER PRIMARY KEY, actor TEXT NOT NULL, action TEXT NOT NULL, 

42 detail TEXT NOT NULL, created_at TEXT NOT NULL DEFAULT (strftime('%Y-%m-%dT%H:%M:%fZ','now'))); 

43 CREATE TABLE IF NOT EXISTS login_attempts ( 

44 username TEXT PRIMARY KEY, failures INTEGER NOT NULL, last_attempt REAL NOT NULL); 

45 ''') 

46 conn.execute('''DELETE FROM academic_records WHERE record_id NOT IN 

47 (SELECT MAX(record_id) FROM academic_records GROUP BY student_id,subject_id,period)''') 

48 conn.execute('CREATE UNIQUE INDEX IF NOT EXISTS record_key ON academic_records(student_id,subject_id,period)')

50def backup_database(destination): 

51 target = Path(destination).resolve() 

52 if target == Path(DB_PATH).resolve(): 

53 raise ValueError('Backup destination must differ from database') 

54 target.parent.mkdir(parents=True, exist_ok=True) 

55 with get_conn() as source: 

56 backup = sqlite3.connect(target)

58 source.backup(backup)
