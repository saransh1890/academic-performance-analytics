50def chart_style(figure, title=None): 

51 figure.update_layout( 

52 title=dict(text=title, font=dict(size=18)) if title else None, 

53 paper_bgcolor='#ffffff', plot_bgcolor='#ffffff', 

54 font=dict(family='Arial, sans-serif', color='#52657A', size=13),

56 legend=dict(title=None, orientation='h', y=-.2, x=0), 

57 hovermode='closest', height=355, 

58 ) 

59 figure.update_yaxes(gridcolor='#EDF1F5', zeroline=False) 

60 figure.update_xaxes(showgrid=False, title=None) 

61 return figure
